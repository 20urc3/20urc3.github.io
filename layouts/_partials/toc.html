{{- /* Left-hand navigation pane listing the article's headings. Built by hand rather than from .TableOfContents because heading levels are inconsistent across posts (astra jumps between h1 and h2) and one post has an empty "## " heading. We collect h1-h3, drop the empty ones, then indent each entry relative to the shallowest level actually present, so every article gets a flat, evenly-indented list regardless of which levels it used. Entries are numbered 1., 1.a, 1.a.1 by that same relative depth. */ -}} {{- $headers := findRE "]*?>(.|\n)+?" .Content -}} {{- /* (id, level, text) triples, empty headings discarded */ -}} {{- $items := slice -}} {{- range $headers -}} {{- $level := int (index (findRE "[1-3]" . 1) 0) -}} {{- $ids := findRE "id=\"[^\"]+\"" . 1 -}} {{- $text := trim (plainify (replaceRE "]*?>((.|\n)+?)" "$1" .)) " \n\t" -}} {{- if and $text $ids -}} {{- $id := replaceRE "(id=\"|\")" "" (index $ids 0) -}} {{- $items = $items | append (dict "id" $id "level" $level "text" $text) -}} {{- end -}} {{- end -}} {{- if ge (len $items) 2 -}} {{- $min := 3 -}} {{- range $items }}{{ if lt .level $min }}{{ $min = .level }}{{ end }}{{ end -}} {{- $letters := split "abcdefghijklmnopqrstuvwxyz" "" -}} {{- $n0 := 0 -}}{{- $n1 := 0 -}}{{- $n2 := 0 -}}

{{ i18n "toc" | default "Table of Contents" }}

    {{- range $items }} {{- $d := sub .level $min }} {{- /* a sub-heading appearing before any top-level one still belongs to a section, so open section 1 rather than numbering it 0.a */ -}} {{- if and (gt $d 0) (eq $n0 0) }}{{ $n0 = 1 }}{{ end }} {{- $label := "" }} {{- if eq $d 0 }} {{- $n0 = add $n0 1 }}{{ $n1 = 0 }}{{ $n2 = 0 }} {{- $label = printf "%d." $n0 }} {{- else }} {{- if eq $d 1 }}{{ $n1 = add $n1 1 }}{{ $n2 = 0 }}{{ end }} {{- /* past 26 sub-sections there are no letters left, so fall back to a number */ -}} {{- $sub := string $n1 }} {{- if and (ge $n1 1) (le $n1 26) }}{{ $sub = index $letters (sub $n1 1) }}{{ end }} {{- if eq $d 1 }} {{- $label = printf "%d.%s" $n0 $sub }} {{- else }} {{- $n2 = add $n2 1 }} {{- $label = printf "%d.%s.%d" $n0 $sub $n2 }} {{- end }} {{- end }}
  • {{ $label }}{{ .text }}

{{- end -}}
